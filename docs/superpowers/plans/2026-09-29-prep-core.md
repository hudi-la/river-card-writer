# 译前备料 · 核心切片 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让作者能在中文旁边自己写英文，并在写的时候从右侧边栏拿到经语料查证过的选词建议。

**Architecture:** 英文以 `string[]`（每段一个字符串）存在中文卡片的 `en` 属性里，所以重排/搬卡时英文自动跟随、不需要第二个编辑器。「过期」由指纹算出来而非存下来。备料由 Supabase Edge Function 薄壳转发给模型，所有拼装与校验逻辑作为纯函数留在 `src/` 里可测；每个候选词再经 Google Books Ngrams 查真实语料频率，分三档显示。

**Tech Stack:** TypeScript · React 19 · TipTap 3 · Zustand · Vitest · Supabase（Postgres + Edge Functions）· Gemini（供应商可换）

**Spec:** `docs/superpowers/specs/2026-09-23-translation-prep-design.md`

## 本计划的范围

这是 spec 的**核心可用切片**，不是全部。交付后作者就能真的用它写英文。

**做**：§3 数据模型、§4 过期、§5 切分合并、§6 条目结构、§10 两栏+边栏、§11 调用通路与查证层、§12 对应测试。

- §3 的 `book.translation`（文体约定 / 术语表 / 参考文本）——有用但不是"能用"的前提。

## Global Constraints

以下是 spec 的项目级要求，每个任务都隐含包含：

- **`en` 字段只装作者自己敲的字。** 模型产出无论哪一档都进 `book.hints`，绝不进 `en`。（§9.1）
- **条目 schema 里不许有任何句子形状的字段。** 没有"例句"字段。没有地方放句子，模型就放不进句子。（§9.2）
- **每个候选必须有非空的 `loses`（它丢了什么）。** 校验层强制，缺失即拒收。（§6）
- **`verify_form` 必须是裸词形**：只写候选自己的词，不加宾语/主语/上下文；原形；固定习语含冠词；≤5 词。（§11.1 的误报陷阱）
- **查证取不到时一律显示为「未查证」**，绝不默认当作已查证。（§11.1）
- **`en.length === 中文段落数`** 恒成立。校验器守在数据入口，不合格的 `en` 不许进文档。（§5）
- **过期是算出来的**，任何地方都不许"记得去打过期标记"。（§4）
- **向后兼容**：老文档没有新属性时解析出默认值，行为零变化——跟现有 `note` / `collapsed` / `cover` 同一个路子。
- 所有注释、UI 文案、提交信息用中文，与现有代码一致。

```ts
// 英文那一半的纯函数底座。刻意不 import 任何 TipTap / React——这一层要能在
// 没有编辑器的环境里单独测，schema 和 CardView 都只是它的调用方。

// "过期"的哨兵值：enSrc 为空串时，任何中文的指纹都匹配不上它，于是 isStale 恒真。
// 用空串而不是造一个假指纹——不确定就是不确定，别装出一个确定值。
export const STALE_SRC = ''

// 归一化后的 FNV-1a。归一化只压空白：改一个空格、末尾多一个换行，不该让作者
// 辛苦写的英文被判过期。除此之外任何字变了都算改了。
export function fingerprint(text: string): string {
  const norm = text.replace(/\s+/g, ' ').trim()
  let h = 0x811c9dc5
  for (let i = 0; i < norm.length; i++) {
    h ^= norm.charCodeAt(i)
    h = Math.imul(h, 0x01000193)
  }
  // 转 36 进制并加前缀：保证结果永远非空，不会意外等于 STALE_SRC。
  return 'f' + (h >>> 0).toString(36)
}

// 过期是算出来的，不是存出来的——所以没有任何地方需要"记得打过期标记"，
// 也就不存在忘记打标记的 bug。
// en 为 null 时返回 false：那是"未写"，跟"过期"是两种不同的状态，UI 上也要分开显示。
export function isStale(en: string[] | null, enSrc: string, zhText: string): boolean {
  if (en == null) return false
  return fingerprint(zhText) !== enSrc
}

// 守在数据入口的校验器，跟现有的 isValidCover 同一个路子：宽松地判断、永不抛出，
// 形状不对就当"不合格"，绝不让它进文档。
// en.length === 段落数 是全功能依赖的不变量（段落样式靠序号从中文继承、切分靠序号
// 对齐），所以段数不符必须拒绝，不能好心地补齐或截断。
export function isValidEn(raw: unknown, paragraphCount: number): raw is string[] | null {
  if (raw === null) return true
  if (!Array.isArray(raw)) return false
  if (raw.length !== paragraphCount) return false
  return raw.every((s) => typeof s === 'string')
}
```

```ts
      en: {
        default: null,
        parseHTML: (el) => {
          const raw = (el as HTMLElement).getAttribute('data-en')
          if (!raw) return null
          try {
            const parsed = JSON.parse(raw)
            const count = (el as HTMLElement).querySelectorAll('p').length
            return isValidEn(parsed, count) ? parsed : null
          } catch {
            return null
          }
        },
        renderHTML: (attrs: { en: string[] | null }) =>
          attrs.en ? { 'data-en': JSON.stringify(attrs.en) } : {},
      },
      // 写英文时那一版中文的指纹。空串 = 显式标记为过期（见 enText.STALE_SRC）。
      enSrc: {
        default: STALE_SRC,
        parseHTML: (el) => (el as HTMLElement).getAttribute('data-en-src') || STALE_SRC,
        renderHTML: (attrs: { enSrc: string }) => (attrs.enSrc ? { 'data-en-src': attrs.enSrc } : {}),
      },
      // 最后一次改动英文的时间（ISO string）。纯信息，不参与任何判断。
      enAt: {
        default: '',
        parseHTML: (el) => (el as HTMLElement).getAttribute('data-en-at') || '',
        renderHTML: (attrs: { enAt: string }) => (attrs.enAt ? { 'data-en-at': attrs.enAt } : {}),
```

```ts
// 把一张卡的若干段中文拼成一个字符串再取指纹。分隔符固定下来，保证
// zhFingerprint 和任何手写的 fingerprint(texts.join(...)) 不会悄悄算出两个结果。
export const ZH_JOIN = '\n'

export function zhFingerprint(texts: string[]): string {
  return fingerprint(texts.join(ZH_JOIN))
}

export type EnState = { en: string[] | null; enSrc: string }

// 切分：中文卡在第 index 个段落处被切成两半。
// 只有"在段落边界下刀"且"英文段数等于中文段数"时才敢对齐切——这两个条件由
// 提示词里"保持段落一一对应"那条约束在多数情况下保证成立。
// 对不齐时不猜：英文整份留给前半张并标过期，后半张当没写过。留一份对不上的
// 英文比清零好（作者的润色不该消失），但必须告诉他它已经不可信了。
export function splitEn(
  en: string[] | null,
  zhTexts: string[],
  index: number,
  atParagraphBoundary: boolean
): { first: EnState; second: EnState } {
  if (en == null) {
    return { first: { en: null, enSrc: STALE_SRC }, second: { en: null, enSrc: STALE_SRC } }
  }
  const alignable =
    atParagraphBoundary && en.length === zhTexts.length && index > 0 && index < en.length
  if (!alignable) {
    return { first: { en, enSrc: STALE_SRC }, second: { en: null, enSrc: STALE_SRC } }
  }
  return {
    first: { en: en.slice(0, index), enSrc: zhFingerprint(zhTexts.slice(0, index)) },
    second: { en: en.slice(index), enSrc: zhFingerprint(zhTexts.slice(index)) },
  }
}

// 合并：译文的拼接就是拼接的译文。接缝处可能读着略生硬，但那是作者一眼看得见、
// 顺手能改的东西，远好过把他的润色清零。
// 只有当一边压根没写过时才标过期——那份英文确实盖不全合并后的中文了。
export function mergeEn(a: string[] | null, b: string[] | null, mergedZhTexts: string[]): EnState {
  if (a == null && b == null) return { en: null, enSrc: STALE_SRC }
  if (a == null || b == null) return { en: (a ?? b) as string[], enSrc: STALE_SRC }
  return { en: [...a, ...b], enSrc: zhFingerprint(mergedZhTexts) }
}
```

```tsx
    zhStyles.push(p.attrs.style || 'body')
  })

  // 脏 en（段数不符、形状不对）一律当没写过。渲染一份跟中文对不上的英文比
  // 什么都不显示更有害。
  const rawEn = node.attrs.en
  const en: string[] | null = isValidEn(rawEn, zhTexts.length) ? rawEn : null
  const stale = isStale(en, node.attrs.enSrc || '', zhTexts.join('\n'))

  // 写回一段英文。只在 blur 时调用——写回会让这个 node view 重渲染，那时焦点
  // 已经离开，所以不会把作者正在打字的光标冲掉。
  function commit(index: number, text: string) {
    const next = en ? [...en] : zhTexts.map(() => '')
    if (next[index] === text) return
    next[index] = text
    // 全空 = 退回"没写过"，而不是存一个全是空串的数组
    const anyText = next.some((s) => s.trim() !== '')
    updateAttributes({
      en: anyText ? next : null,
      enSrc: anyText ? zhFingerprint(zhTexts) : '',
      enAt: anyText ? new Date().toISOString() : '',
    })
  }

  return (
    <NodeViewWrapper
      as="section"
      className="card"
      data-card-id={cardId || undefined}
      data-color={color || undefined}
      data-en-stale={stale ? 'true' : undefined}
      style={style}
    >
      <span className="card-drag-handle" aria-hidden="true">
        <GripVertical size={14} strokeWidth={1.75} />
      </span>
      <NodeViewContent className="card-body" />
      <aside className="card-en" contentEditable={false}>
        {stale ? <p className="card-en-stale-note">中文改过，这段英文已过期</p> : null}
        {zhTexts.map((_, i) => (
          <div
            key={i}
            className="card-en-p"
            data-style={zhStyles[i]}
            contentEditable
            suppressContentEditableWarning
            role="textbox"
            aria-label={`第 ${i + 1} 段的英文`}
            onBlur={(e) => commit(i, e.currentTarget.textContent ?? '')}
          >
            {en ? en[i] : ''}
          </div>
        ))}
      </aside>
    </NodeViewWrapper>
  )
}
```

**注意**：`<aside>` 上的 `contentEditable={false}` 是必须的——它把英文栏从 ProseMirror 的可编辑区里摘出来，否则 PM 会试图把里面的输入当成文档编辑。内层每个 `.card-en-p` 再单独开 `contentEditable`。

`src/prep/parse.ts`：

```ts
import type { Candidate, Genre, PrepResult, Target } from './types'

// 「我没有缺点」的各种说法。模型倾向于用这些糊过去，一律当成没填。
const EMPTY_LOSES = new Set(['', '无', '没有', 'n/a', 'na', 'none', '—', '-', '不适用'])

// 判断一个字段里是不是塞进了整句英文。启发式：连续 8 个以上的英文单词，
// 或者出现句末标点后又接着大写开头——那就是句子，不是词典条目该有的东西。
// 宁可偶尔误伤（作者会看到拒收原因），也不能让成句从侧门溜进来。
function looksLikeSentence(s: string): boolean {
  const words = s.match(/[A-Za-z][A-Za-z'-]*/g) ?? []
  if (words.length >= 8) return true
  return /[.!?]\s+[A-Z]/.test(s)
}

function wordCount(s: string): number {
  return s.trim().split(/\s+/).filter(Boolean).length
}

type Fail = { ok: false; error: string }
type Ok = { ok: true; value: PrepResult }

function fail(error: string): Fail {
  return { ok: false, error }
}

function parseCandidate(raw: any, where: string): Candidate | Fail {
  if (!raw || typeof raw !== 'object') return fail(`${where}：候选不是对象`)
  const loses = typeof raw.loses === 'string' ? raw.loses : ''
  if (EMPTY_LOSES.has(loses.trim().toLowerCase())) {
    return fail(`${where}：候选「${raw.en}」缺 loses（它丢了什么）。没有任何候选可以只有优点。`)
  }
  const verifyForm = typeof raw.verify_form === 'string' ? raw.verify_form.trim() : ''
  if (!verifyForm) return fail(`${where}：候选「${raw.en}」缺 verify_form`)
  if (wordCount(verifyForm) > 5) {
    return fail(`${where}：verify_form「${verifyForm}」超过 5 个词，语料里查不到，会误报成未查证`)
  }
  for (const [field, v] of [['nuance', raw.nuance], ['loses', loses], ['why_fits', raw.why_fits]] as const) {
    if (typeof v === 'string' && looksLikeSentence(v)) {
      return fail(`${where}：${field} 里出现了整句英文。条目里不该有句子。`)
    }
  }
  return {
    en: String(raw.en ?? ''),
    verifyForm,
    register: String(raw.register ?? ''),
    nuance: String(raw.nuance ?? ''),
    whyFits: String(raw.why_fits ?? ''),
    loses,
  }
}

function parseTarget(raw: any, genre: Genre, i: number): Target | Fail {
  const where = `第 ${i + 1} 条靶心`
  if (!raw || typeof raw !== 'object') return fail(`${where}：不是对象`)
  // 承重句只存在于议论/学术文体。叙事文没有论点句——模型曾经借这个分类
  // 表达"这条最重要"，那是 priority 的活。
  if (raw.category === '承重句' && genre !== 'expository') {
    return fail(`${where}：文体是 ${genre}，不该出现「承重句」分类（那是议论文专属；重要程度用 priority）`)
  }
  const rawCands = Array.isArray(raw.candidates) ? raw.candidates : []
  if (rawCands.length === 0) return fail(`${where}：一个候选都没有`)
  const candidates: Candidate[] = []
  for (const rc of rawCands) {
    const c = parseCandidate(rc, where)
    if ('ok' in c) return c
    candidates.push(c)
  }
  const ft = raw.flat_translation ?? {}
  return {
    zh: String(raw.zh ?? ''),
    zhQuote: String(raw.zh_quote ?? ''),
    category: String(raw.category ?? ''),
    priority: Number.isFinite(raw.priority) ? Number(raw.priority) : 3,
    whyZhIsPrecise: String(raw.why_zh_is_precise ?? ''),
    flatTranslation: { text: String(ft.text ?? ''), whyItDies: String(ft.why_it_dies ?? '') },
    candidates,
    noSingleWord: raw.no_single_word === true,
    family: String(raw.family ?? ''),
  }
}

// 永不抛出：任何形状问题都返回 { ok: false, error }，由调用方决定怎么告诉作者。
export function parsePrepResponse(raw: string): Ok | Fail {
  let json: any
  try {
    json = JSON.parse(raw)
  } catch {
    return fail('模型没有返回合法 JSON')
  }
  if (!json || typeof json !== 'object') return fail('返回值不是对象')
  const genre: Genre = ['narrative', 'expository', 'other'].includes(json.genre) ? json.genre : 'other'
  const rawTargets = Array.isArray(json.targets) ? json.targets : []
  if (rawTargets.length === 0) return fail('一条靶心都没有')
  const targets: Target[] = []
  for (let i = 0; i < rawTargets.length; i++) {
    const t = parseTarget(rawTargets[i], genre, i)
    if ('ok' in t) return t
    targets.push(t)
  }
  targets.sort((a, b) => a.priority - b.priority)
  return { ok: true, value: { genre, genreReason: String(json.genre_reason ?? ''), targets } }
}
```

```ts
// 语料查证层。它存在的唯一理由：LLM 会造出流利但没人真那么说的英文搭配
// （真跑产出过 cerebral haughtiness、architecting the initial conditions），
// 而对一个"帮人选词"的工具，这是最坏的失败——作者信了、写进稿子、交出去。
//
// 对策不是要求模型自觉，是拿真实语料统计去查。统计里生造的词没地方躲。
//
// 能保证的边界要说清楚：保证不了模型不再生造词，也保证不了捞全每个靶心。
// 能保证的是——未查证的词绝不和已查证的词长得一样。

export type Tier = 'ok' | 'rare' | 'unverified'

// 门槛是相对的，重点是分档本身。这个值来自实测：swat down 1.5e-9（确实罕用）
// vs hem in 7.1e-8（正常书面词），1e-8 把两者分开。
export const RARE_BELOW = 1e-8

// null（查不到）→ unverified。CRITICAL：绝不默认当成已查证。
// 0 → rare 而不是 unverified：那是"查到了，只是极少"，跟"查不到"是两件事。
export function tierOf(freq: number | null): Tier {
  if (freq == null) return 'unverified'
  return freq < RARE_BELOW ? 'rare' : 'ok'
}

export type VerifyFn = (form: string) => Promise<number | null>

// Google Books Ngrams。非官方端点，所以取不到是常态而不是异常——调用方必须
// 有降级路径。
//
// 三个盲区必须记在这里，否则以后有人会把它当成权威：
//   1. 最多 5 个词，更长的短语根本查不了。
//   2. 语料截止 2019，系统性低估新词（people-pleasing 只有 2.05e-8，
//      明显低于它真实的当代使用度）。一个词可能仅仅因为"新"就被标可疑。
//   3. 只有书面语料。对本用途是优点（学术/文学语域覆盖得好），但口语测不到。
export async function fetchNgramFreq(form: string): Promise<number | null> {
  const url =
    'https://books.google.com/ngrams/json?content=' +
    encodeURIComponent(form) +
    '&year_start=1980&year_end=2019&corpus=en-2019&smoothing=3'
  const r = await fetch(url)
  if (!r.ok) return null
  const j = (await r.json()) as Array<{ timeseries: number[] }>
  if (!Array.isArray(j) || j.length === 0) return null
  const ts = j[0]?.timeseries
  if (!Array.isArray(ts) || ts.length === 0) return null
  const recent = ts.slice(-10)
  return recent.reduce((a, b) => a + b, 0) / recent.length
}

// 一个词的历史语料频率不会变，所以缓存可以永久有效。
// 只缓存成功的结果——取数失败不写缓存，否则一次网络抖动会把一个好词
// 永久钉在"未查证"上。
export function createVerifier(fetchFreq: VerifyFn = fetchNgramFreq) {
  const cache = new Map<string, Tier>()

  return async function verify(forms: string[]): Promise<Map<string, Tier>> {
    const unique = Array.from(new Set(forms.filter((f) => f.trim() !== '')))
    const missing = unique.filter((f) => !cache.has(f))

    await Promise.all(
      missing.map(async (form) => {
        try {
          cache.set(form, tierOf(await fetchFreq(form)))
        } catch {
          // 取不到就降级成未查证，绝不让整次备料失败；也不写缓存，下次再试。
        }
      })
    )

    const out = new Map<string, Tier>()
    for (const f of unique) out.set(f, cache.get(f) ?? 'unverified')
    return out
  }
}
```

```ts
// 拼提示词和校验返回值全在 src/prep/ 里作为纯函数——那些要能用 vitest 测，
// 而网络这一层薄到不值得测。

// 为什么走 Edge Function 而不是浏览器直连：站点公开部署在 card-writer.pages.dev，

// 部署：supabase functions deploy prep
// 设置 key：supabase secrets set GEMINI_API_KEY=...
const MODEL = Deno.env.get('PREP_MODEL') ?? 'gemini-3.5-flash'

Deno.serve(async (req) => {
  if (req.method !== 'POST') return new Response('method not allowed', { status: 405 })
  const key = Deno.env.get('GEMINI_API_KEY')
  if (!key) return new Response(JSON.stringify({ error: '服务端没有配置 GEMINI_API_KEY' }), { status: 500 })

  let body: { system?: string; user?: string }
  try {
    body = await req.json()
  } catch {
    return new Response(JSON.stringify({ error: '请求体不是合法 JSON' }), { status: 400 })
  }
  if (!body.system || !body.user) {
    return new Response(JSON.stringify({ error: '缺 system 或 user' }), { status: 400 })
  }

  const r = await fetch(
    `https://generativelanguage.googleapis.com/v1beta/models/${MODEL}:generateContent?key=${key}`,
    {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({
        systemInstruction: { parts: [{ text: body.system }] },
        contents: [{ role: 'user', parts: [{ text: body.user }] }],
        // 输出是清单不是稿子，又短又便宜，所以不需要流式。
        generationConfig: { responseMimeType: 'application/json', temperature: 1 },
      }),
    }
  )
  const j = await r.json()
  if (!r.ok) {
    return new Response(JSON.stringify({ error: `模型返回 ${r.status}`, detail: j }), { status: 502 })
  }
  const text = j?.candidates?.[0]?.content?.parts?.map((p: { text?: string }) => p.text ?? '').join('') ?? ''
  return new Response(JSON.stringify({ text }), { headers: { 'content-type': 'application/json' } })
```

```ts
import { supabase } from '../lib/supabase'
import { buildPrepRequest } from './prompt'
import { parsePrepResponse } from './parse'
import type { PrepResult } from './types'

// invoke 抽成可注入的依赖：测试不能真的打到线上 Edge Function。
// 跟 triggerSave / startSync 一样的"默认走真实模块、测试可注入"约定。
export type InvokeFn = (
  name: string,
  opts: { body: unknown }
) => Promise<{ data: { text?: string } | null; error: { message: string } | null }>

const defaultInvoke: InvokeFn = (name, opts) =>
  (supabase as unknown as { functions: { invoke: InvokeFn } }).functions.invoke(name, opts)

// 永不抛出：任何失败都变成 { ok: false, error }，由 UI 决定怎么告诉作者。
export async function requestPrep(
  zhTexts: string[],
  deps: { invoke?: InvokeFn } = {}
): Promise<{ ok: true; value: PrepResult } | { ok: false; error: string }> {
  const invoke = deps.invoke ?? defaultInvoke
  try {
    const { system, user } = buildPrepRequest(zhTexts)
    const { data, error } = await invoke('prep', { body: { system, user } })
    if (error) return { ok: false, error: error.message }
    if (!data?.text) return { ok: false, error: '服务端没有返回内容' }
    return parsePrepResponse(data.text)
  } catch (e) {
    return { ok: false, error: e instanceof Error ? e.message : String(e) }
  }
}
```

```tsx
const TIER_LABEL: Record<Tier, string> = { ok: '已查证', rare: '罕用', unverified: '未查证' }

function TierBadge({ tier }: { tier: Tier }) {
  return (
    <span className="prep-tier" data-tier={tier}>
      {TIER_LABEL[tier]}
      {tier === 'unverified' ? ' · 语料里查不到' : null}
    </span>
  )
}

// 右侧固定边栏：显示当前那张卡的备料。位置固定，眼睛不用找；提词再长也
// 撑不散河流（这是"两栏 + 一条边栏"而不是真三栏的理由——三栏等宽后中文
// 英文都只剩三四十字宽，而那两栏才是作者反复比对的主力）。
export function PrepSidebar({
  targets,
  genre,
  tiers,
  aiPolicy,
}: {
  targets: Target[]
  genre: Genre
  tiers: Map<string, Tier>

          ) : null}
```

```tsx
  const verify = useMemo(() => createVerifier(), [])
  const [tiers, setTiers] = useState<Map<string, Tier>>(new Map())
  const [prepping, setPrepping] = useState(false)

  async function prepChapter() {
    const book = activeBook()
    if (!book) return
    setPrepping(true)
    // 一张卡一次请求，带着它自己的段落。前后邻居上下文留给后续计划。
    const cards = (book.doc as { content?: Array<{ attrs?: { cardId?: string }; content?: Array<{ content?: Array<{ text?: string }> }> }> }).content ?? []
    for (const card of cards) {
      const cardId = card.attrs?.cardId
      if (!cardId) continue
      const zhTexts = (card.content ?? []).map((p) => (p.content ?? []).map((t) => t.text ?? '').join(''))
      if (zhTexts.join('').trim() === '') continue
      const r = await requestPrep(zhTexts)
      if (!r.ok) {
        showToast({ text: `备料失败：${r.error}`, kind: 'error', durationMs: 5000 })
        continue
      }
      setCardHints(cardId, r.value)
      // 查证每个候选的裸词形。取不到一律显示未查证，绝不默认当成已查证。
      const forms = r.value.targets.flatMap((t) => t.candidates.map((c) => c.verifyForm))
      const m = await verify(forms)
      setTiers((prev) => new Map([...prev, ...m]))
    }
    setPrepping(false)
  }
```
